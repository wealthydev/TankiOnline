// =========================================================================================
//  COFFEE KERNEL
//  Loaded by @require, ahead of script.js. Everything here is infrastructure: no feature
//  logic, no game knowledge beyond how the bundle names things.
//
//  Five parts, in dependency order:
//
//    1. LOG      the real console, captured before the game's own logging is silenced
//    2. BUS      a small event emitter - how features talk to each other
//    3. CLOCK    ONE scheduler. Every periodic and per-frame job in the script runs here
//    4. ACCESS   reading the bundle's mangled objects by position instead of by name
//    5. REFLECT  the symbol table built from the bundle's own toString methods
//
//  VERSIONED ON PURPOSE. This file is @require'd from GitHub and Tampermonkey caches it,
//  so script.js can be newer than the kernel the browser actually has. script.js checks
//  KERNEL.version at boot and says so plainly rather than failing in pieces. Bump MINOR
//  when adding, MAJOR when changing or removing anything below.
// =========================================================================================

window.KERNEL = { version: 2, major: 2, minor: 0 };

// ---- 1. LOG ----------------------------------------------------------------------------
// The game logs heavily to the console, which buries ours. The real function is taken first,
// then the game's is silenced - order matters, and this must stay at the top of the file.
window.log = console.log.bind(console);
console.info = console.log = () => {};

// ---- 2. BUS ----------------------------------------------------------------------------
// Features announce facts ("the battle ended", "a chat line arrived") and whoever cares
// listens. It replaces the pattern this script grew up with - one feature polling a
// window.* flag that another feature happens to set - which is why so much of it ran every
// frame to notice something that happens twice a match.
//
// emit() must be cheap enough for the hot path: it allocates nothing, and takes no copy of
// the listener array. Removal during a dispatch is handled by tombstoning rather than
// splicing, so a listener that unsubscribes itself cannot make the loop skip its neighbour.
window.Bus = (() => {
    const map = new Map();               // type -> array of fns (may contain null holes)
    const dirty = new Set();             // types with holes to compact after dispatch
    let depth = 0;

    const compact = () => {
        for (const type of dirty) {
            const arr = map.get(type);
            if (!arr) continue;
            let w = 0;
            for (let r = 0; r < arr.length; r++) if (arr[r]) arr[w++] = arr[r];
            arr.length = w;
            if (!w) map.delete(type);
        }
        dirty.clear();
    };

    const off = (type, fn) => {
        const arr = map.get(type);
        if (!arr) return;
        const i = arr.indexOf(fn);
        if (i < 0) return;
        arr[i] = null;
        dirty.add(type);
        if (!depth) compact();
    };

    return {
        /** Listen. Returns the unsubscribe function, so callers need not keep the fn. */
        on(type, fn) {
            let arr = map.get(type);
            if (!arr) map.set(type, (arr = []));
            arr.push(fn);
            return () => off(type, fn);
        },
        /** Listen for the next one only. */
        once(type, fn) {
            const un = this.on(type, (p) => { un(); fn(p); });
            return un;
        },
        off,
        /**
         * Announce. One misbehaving listener is reported and skipped - it never stops the
         * others, and never propagates into whatever emitted.
         */
        emit(type, payload) {
            const arr = map.get(type);
            if (!arr) return;
            depth++;
            for (let i = 0; i < arr.length; i++) {
                const fn = arr[i];
                if (!fn) continue;
                try { fn(payload); }
                catch (e) { window.log(`[bus] listener for "${type}" threw:`, e.message); }
            }
            if (!--depth) compact();
        },
        /** For the console: which events have listeners, and how many. */
        listeners() {
            const out = {};
            for (const [t, a] of map) out[t] = a.filter(Boolean).length;
            return out;
        },
    };
})();

// ---- 3. CLOCK --------------------------------------------------------------------------
// ONE requestAnimationFrame loop and one timer, for the whole script.
//
// The script this replaces had three separate rAF loops and eight setIntervals, several of
// which ran flat out to do nothing: a fullscreen canvas cleared every frame with every
// feature switched off, a DOM query every 7ms for a question answered once at startup.
//
// Three things make that hard to reintroduce here:
//
//   * `when` - a task with a false `when` costs one predicate call and is not entered. This
//     is where an idle feature's cost goes to nearly zero, and it is a property of the task
//     rather than something each feature has to remember to check.
//   * `hz` - a ceiling. Most work in a game overlay does not need 60 updates a second, and
//     saying so is one number rather than a hand-rolled Date.now() comparison per feature.
//   * `stats()` - per-task timing. When the frame rate drops, the answer should be a
//     measurement, not a guess. This is the difference between "something in here is slow"
//     and "this task is 4ms".
//
// A throwing task is disabled after it has reported itself a few times, rather than being
// left to throw every frame forever. In the old arrangement a single bad frame in the
// overlay's loop killed the whole overlay silently, because the throw escaped the rAF
// callback and nothing rescheduled it.
window.Clock = (() => {
    const frames = [];        // per-frame tasks, kept sorted by order
    const timers = new Map(); // name -> {id} for the background ones
    let running = false;
    let now = 0;

    const ERR_LIMIT = 5;

    const mkTask = (name, fn, opt) => ({
        name,
        fn,
        when: opt.when || null,
        minMs: opt.hz ? 1000 / opt.hz : 0,
        order: opt.order == null ? 100 : opt.order,
        last: 0,
        calls: 0,
        ms: 0,            // exponential moving average, milliseconds
        peak: 0,
        errors: 0,
        off: false,
    });

    const runTask = (t, ts) => {
        if (t.off) return;
        if (t.minMs && ts - t.last < t.minMs) return;
        if (t.when) {
            let want;
            try { want = t.when(); }
            catch (e) { want = false; }
            if (!want) return;
        }
        t.last = ts;
        const t0 = performance.now();
        try {
            t.fn(ts);
        } catch (e) {
            if (++t.errors <= ERR_LIMIT) {
                window.log(`[clock] "${t.name}" threw:`, e.message);
                if (t.errors === ERR_LIMIT) {
                    t.off = true;
                    window.log(`[clock] "${t.name}" disabled after ${ERR_LIMIT} errors`);
                }
            }
            return;
        }
        const dt = performance.now() - t0;
        t.calls++;
        t.ms += (dt - t.ms) * 0.1;
        if (dt > t.peak) t.peak = dt;
    };

    const loop = (ts) => {
        now = ts;
        for (let i = 0; i < frames.length; i++) runTask(frames[i], ts);
        requestAnimationFrame(loop);
    };

    const start = () => {
        if (running) return;
        running = true;
        requestAnimationFrame(loop);
    };

    return {
        /**
         * Run every frame. `order` fixes the sequence (lower first), `hz` caps the rate,
         * `when` decides - per frame, cheaply - whether it runs at all.
         */
        frame(name, fn, opt = {}) {
            const t = mkTask(name, fn, opt);
            frames.push(t);
            frames.sort((a, b) => a.order - b.order);
            start();
            return () => {
                const i = frames.indexOf(t);
                if (i >= 0) frames.splice(i, 1);
            };
        },
        /**
         * Run every ms. Driven off the same rAF by default, so it batches with the frame
         * and stops while the tab is hidden. `background: true` gives it a real timer
         * instead, for the few jobs that must keep going when the tab is not visible -
         * a socket heartbeat, for instance.
         */
        every(ms, name, fn, opt = {}) {
            if (opt.background) {
                const id = setInterval(() => {
                    if (opt.when) { let w; try { w = opt.when(); } catch { w = false; } if (!w) return; }
                    try { fn(); } catch (e) { window.log(`[clock] "${name}" threw:`, e.message); }
                }, ms);
                timers.set(name, { id });
                return () => { clearInterval(id); timers.delete(name); };
            }
            return this.frame(name, fn, Object.assign({}, opt, { hz: 1000 / ms }));
        },
        /** Run once, after ms. */
        after(ms, name, fn) {
            const id = setTimeout(() => {
                timers.delete(name);
                try { fn(); } catch (e) { window.log(`[clock] "${name}" threw:`, e.message); }
            }, ms);
            timers.set(name, { id });
            return () => { clearTimeout(id); timers.delete(name); };
        },
        /** The timestamp of the frame in progress. */
        now: () => now,
        /**
         * What everything costs, worst first. This is the tool for the next time the frame
         * rate sags: `Clock.stats()` in the console names the task instead of leaving it to
         * be guessed at.
         */
        stats() {
            return frames
                .map((t) => ({
                    task: t.name,
                    avg_ms: +t.ms.toFixed(3),
                    peak_ms: +t.peak.toFixed(2),
                    calls: t.calls,
                    hz_cap: t.minMs ? Math.round(1000 / t.minMs) : 60,
                    idle: !!(t.when && !safeWhen(t)),
                    off: t.off,
                }))
                .sort((a, b) => b.avg_ms - a.avg_ms);
        },
        /** Total average milliseconds per frame across every task. */
        budget() {
            return +frames.reduce((s, t) => s + t.ms, 0).toFixed(3);
        },
    };

    function safeWhen(t) {
        try { return t.when(); } catch { return false; }
    }
})();

// ---- 4. ACCESS -------------------------------------------------------------------------
// The bundle's property names are mangled and change every build, so the script addresses
// the engine's objects BY POSITION. That is the whole reason this section exists.

// Object.values() builds an array of every value in an object to read one slot. Doing that
// per tank per frame was the largest single source of garbage in the script, so the key
// order - which never changes for a given object once it exists - is cached against the
// object itself. A WeakMap means a tank leaving the battle takes its entry with it.
window._keysOf = new WeakMap();
window.keysOf = (o) => {
    let k = window._keysOf.get(o);
    if (k === undefined) { k = Object.keys(o); window._keysOf.set(o, k); }
    return k;
};

// One positional hop. This mirrors byIndex's loop body exactly, including the parts that are
// easy to get wrong: a falsy value passes straight through instead of becoming undefined, a
// string indexes by character (Object.values of a string is its characters), and a number or
// boolean reads as undefined (Object.values of either is empty).
window._step = (o, i) => {
    if (!o) return o;
    if (typeof o === "string") return o[i];
    if (typeof o !== "object") return undefined;
    return o[window.keysOf(o)[i]];
};

// byIndex's own guard is a bare `return`, so a falsy ARGUMENT gives undefined - which is not
// the same as the pass-through a falsy intermediate gets above.
/** byIndex(o, i) with no allocation. */
window.at = (o, i) => (o ? window._step(o, i) : undefined);
/** byIndex(o, i, j) with no allocation - the two-level form, which is the common one. */
window.at2 = (o, i, j) => (o ? window._step(window._step(o, i), j) : undefined);

/**
 * The original variadic accessor. Kept for cold paths and for console use, where an extra
 * array costs nothing and the variadic form is more convenient than nesting at().
 */
window.byIndex = function (obj) {
    if (!obj) return;
    let res = obj;
    for (let i = 1; i < arguments.length; i++) {
        const index = arguments[i];
        if (typeof index === "number" && res) res = Object.values(res);
        if (res) res = res[index];
    }
    return res;
};

window.setByIndex = function (obj) {
    if (!obj) return;
    const value = arguments[arguments.length - 1];
    let res = obj;
    for (let i = 1; i < arguments.length - 1; i++) {
        const index = arguments[i];
        if (i === arguments.length - 2) {
            res[Object.keys(res)[index]] = value;
        } else {
            if (typeof index === "number") res = Object.values(res);
            res = res[index];
        }
    }
    return res;
};

/** Walk by the NAMES the symbol table recovered, rather than by position. */
window.byPath = function (obj, path) {
    if (!obj) return;
    try {
        const parts = path.split(".");
        let res = obj;
        for (const want of parts) {
            for (const key of Object.keys(res)) {
                if (window._collection[window.devanilla(key)] === want) {
                    res = res[key];
                } else if (obj[key] && typeof obj[key] === "object" && !Array.isArray(obj[key])) {
                    for (const c of Object.keys(obj[key])) {
                        if (window._parents[window.devanilla(c)] === want) res = obj[key];
                    }
                }
            }
        }
        return res;
    } catch (e) { /* the shape moved; the caller handles undefined */ }
    return null;
};

window.setByPath = function (obj, path, value) {
    if (!obj) return;
    const parts = path.split(".");
    let res = obj;
    for (let i = 0; i < parts.length; i++) {
        const want = parts[i];
        const end = i === parts.length - 1;
        for (const key of Object.keys(res)) {
            if (window._collection[window.devanilla(key)] === want) {
                if (end) res[key] = value;
                res = res[key];
            } else if (obj[key] && typeof obj[key] === "object" && !Array.isArray(obj[key])) {
                for (const c of Object.keys(obj[key])) {
                    if (window._parents[window.devanilla(c)] === want) res = obj[key];
                }
            }
        }
    }
    return res;
};

/** Copy one engine object's values into another, slot by slot. */
window.copyItem = (item, item2) => {
    Object.values(item).forEach((_, index) => {
        window.setByIndex(item, index, window.byIndex(item2, index));
    });
};

/** Distance between two engine vectors, whatever their field names are. */
window.dist = (a, b) => {
    a = Object.values(a); b = Object.values(b);
    const x = b[0] - a[0], y = b[1] - a[1], z = b[2] - a[2];
    return Math.sqrt(x * x + y * y + z * z);
};

/**
 * Define a property on Object.prototype so every object in the engine is observed through
 * it. This is how the script hooks values it cannot otherwise reach: the bundle assigns to
 * a mangled field, and the setter here sees it.
 *
 * The real value is kept under `_name_`, which is why devanilla() exists.
 */
window.apply = (property, setter, getter) => property && Object.defineProperty(
    Object.prototype, property, {
        get() {
            return getter
                ? getter.call(this, `_${property}_`, this[`_${property}_`])
                : this[`_${property}_`];
        },
        set(value) { this[`_${property}_`] = setter ? setter.call(this, value) : value; },
        configurable: true,
    });

/** Strip the underscores apply() wraps a hooked field in. */
window.devanilla = (str) =>
    (str.startsWith("_") && str.endsWith("_") ? str.slice(1, -1) : str);

window.sleep = (ms) => new Promise((r) => setTimeout(r, ms));

/** Find the game's own main bundle URL, waiting for it to appear. */
window.getMain = async (url) => {
    while (!(url = [...document.scripts].find((s) => s.src.includes("/static/js/"))?.src)) {
        await window.sleep(100);
    }
    return url.split("com")[1];
};

/** Split, find, extract - used against bundle text. */
window.extract = (text, sep, match, take) => {
    let out;
    for (const part of text.split(sep)) if (match(part) && (out = take(part))) break;
    return out;
};

// ---- 5. REFLECT ------------------------------------------------------------------------
// THE SYMBOL TABLE.
//
// Kotlin/JS mangles every field name, and the mangling changes every build - so nothing in
// this script may hardcode one. What it can rely on is that the compiler emits a toString()
// for its data classes, and those print the ORIGINAL names next to the mangled fields:
//
//     "UserLabelData(userId=" + this.k1fx_1 + ", uid=" + ...
//
// setup() reads every one of those in the bundle and builds three maps:
//
//     _collection   mangled field  -> real name        (the main one)
//     _parents      mangled field  -> owning class     (disambiguates common names)
//     _callable     mangled class  -> callableName     (the engine's own registry)
//
// Everything else in the script looks names up here, which is what makes a weekly bundle
// update a non-event.
window.setup = (str) => {
    const introduce = (type, key, value) => {
        if (!window[type]) { window[type] = {}; window[`_${type}`] = {}; }
        if (!key || !value || value.includes(".") || window[`_${type}`][value]) return;
        window[`_${type}`][value] = key;
        window[type][key] = window[type][key] ? `${window[type][key]},${value}` : value;
    };

    // The several shapes a printed field takes across the compiler's output.
    const FIELD_RES = [
        /(\w+)="\+this\.(\w+)\+"/g,
        /(\w+)="\+this\.(\w+)(?:\(\)|\w*)\+/g,
        /(\w+)\s*=\s*"\+this\.(\w+)/g,
        /(\w+)="\+Td\(this\.(\w+)\)/g,
        /(\w+)\s*=\s*(?:"\+)?\w+\(this\.(\w+)\)/g,
        /(\w+):\s*"\+this\.(\w+)/g,
        /(\w+)="\+new\s+iR\(this\.(\w+)\)/g,
    ];

    // Which class does a printed field belong to? Taken from the class name the toString
    // opens with, so a field called "name" on two classes stays distinguishable.
    //
    // ============================ DO NOT ADD A THIRD FORM HERE ============================
    // The original had THREE. The middle one used matchAll() with a NON-GLOBAL regex, which
    // throws TypeError, straight into the catch below - so it never ran, not once, on any
    // build. It is omitted rather than repaired: every one of the 61 patch boxes resolves its
    // names through the table this function builds, and they were all written against the
    // table as it actually is. Giving that form a /g and letting it run changes how
    // BattleMap, Material and Prop resolve, and silently moves anchors out from under boxes
    // that currently match. Measured, not guessed: restoring it added 325 parent entries and
    // changed five classes.
    // ======================================================================================
    const CLASS_FORMS = [
        {
            cls: /"(\w+)\((.*?)\)"/,
            param: /(\w+)="\+(?:\w+\()?(.+?)(?:\))?\+/g,
            clean: (v) => v.replace(/.*?([a-zA-Z0-9_]+)$/, "$1"),
        },
        { cls: /"(\w+)\s*\["/, param: /(\w+)\s*=\s*"\+\w+\(this\.([a-zA-Z0-9_]+)\)/g, clean: (v) => v },
    ];

    const parents = (line) => {
        for (const form of CLASS_FORMS) {
            try {
                const m = line.match(form.cls);
                if (!m) continue;
                const pairs = [...line.matchAll(form.param)].map((x) => [x[1], form.clean(x[2])]);
                // Only the FIRST form that yields any pairs wins; a form that matches the
                // class but finds no fields falls through to the next one.
                if (pairs.length) {
                    for (const [, field] of pairs) introduce("parents", m[1], field);
                    return;
                }
            } catch (e) { /* this form does not apply to this build */ }
        }
    };

    const toStrings = (line) => {
        if (!line.includes('"')) return;
        parents(line);
        for (const re of FIELD_RES) {
            for (const m of line.matchAll(re)) introduce("collection", m[1], m[2]);
        }
        // ===================== THE NUMBERED FALLBACK IS GONE ON PURPOSE =====================
        // The original followed this with a loop meant to register any remaining fields under
        // `classname_1`, `classname_2`, ... It destructured `[c, o, i]` out of a two-element
        // match array, so `i` was ALWAYS undefined, and introduce() rejects an undefined
        // value at its first guard. The loop therefore did nothing whatsoever.
        //
        // Reinstating it is not a fix, it is a behaviour change: it adds 792 entries to
        // _collection and rebinds names the boxes already resolve (`resources` moved from
        // `rif,oif_1` to `oif_1`, `parameters` from `l2a,tok_1` to `tok_1`). Both numbers are
        // measured against this bundle.
        //
        // It also computed `line.match(/"([^(]+)/)[1]`, which throws on a line whose only
        // quote is its last character - an uncaught throw that would abort the entire symbol
        // table. Dropping the loop removes that hazard as well.
        // ====================================================================================
    };

    // The engine keeps its own name for many classes in a `callableName` field; where that
    // exists it beats anything inferred.
    const callableNames = (str_) => {
        const re = /function\s+([^(]+)|callableName\s*=\s*"([^"]+)"/g;
        let fn = null, m;
        while ((m = re.exec(str_)) !== null) {
            if (m[1]) fn = m[1];
            else if (m[2] && fn) introduce("callable", m[2], fn);
        }
        // A handful of fields are reached through a getter rather than printed, so they are
        // picked out by the names the engine happens to spell in full.
        const SPELLED = ["targetingSystem", "angularVelocity", "chatModeratorLevel", "linksWhiteList"];
        for (const want of SPELLED) {
            try {
                const before = str_.split(`"${want}")`)[0];
                const getter = before.split("if(").pop().split(";")[1].split("(")[0];
                for (const v of ["n", "t", "i"]) {
                    str_.split(`if(null!=${v})return ${v};${getter}("`).slice(1)
                        .forEach((part, i, all) => {
                            const name = part.split('"')[0];
                            const field = all[i - 1]?.slice(-20).split(`${v}=`)[1]
                                ?.split(".")[1]?.split(";")[0];
                            introduce("collection", name, field);
                        });
                }
            } catch (e) { /* that one is not in this build */ }
        }
    };

    const chunks = str.split(`).toString=function(){`);
    chunks.shift();
    for (const chunk of chunks) toStrings(chunk.split("}")[0]);
    callableNames(str);
};

/** Rewrite an object's mangled keys to their real names - for reading in the console. */
window.getNames = (arr, maxDepth = 2) => {
    const walk = (obj, depth) => {
        if (depth > maxDepth) return obj;
        if (Array.isArray(obj)) return obj.map((v) => walk(v, depth + 1));
        if (typeof obj !== "object" || obj === null) return obj;
        const out = {};
        for (const key of Object.keys(obj)) {
            let name = window._collection[window.devanilla(key)]
                || window._callable[obj?.__proto__?.constructor?.name]
                || window.byProto(obj) || key;
            if (!window.devanilla(key) && obj[key] && typeof obj[key] === "object"
                && !Array.isArray(obj[key])) {
                for (const c of Object.keys(obj[key])) {
                    const p = window._parents[window.devanilla(c)];
                    if (p) name = p;
                }
            }
            out[name] = walk(obj[key], depth + 1);
        }
        return out;
    };
    return walk(Array.isArray(arr) ? [...arr] : { ...arr }, 0);
};

/** The engine's own name for an object, read off its prototype's callableName fields. */
window.byProto = (obj) => {
    const proto = obj?.__proto__;
    if (!proto) return null;
    let name = null;
    for (const key of Object.keys(proto)) {
        const s = String(proto[key]);
        const got = s.includes(`callableName="`)
            ? s.split(`callableName="`)[1].split(`"`)[0] : null;
        if (got) name = `${name ? `${name}_` : ""}${got}`;
    }
    return name;
};
