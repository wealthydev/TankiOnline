function _rc4(key, data) {
    const S = new Uint8Array(256);
    for (let i = 0; i < 256; i++) S[i] = i;
    let j = 0;
    for (let i = 0; i < 256; i++) {
        j = (j + S[i] + key.charCodeAt(i % key.length)) & 255;
        const t = S[i]; S[i] = S[j]; S[j] = t;
    }
    let a = 0, b = 0;
    for (let k = 0; k < 256; k++) {
        a = (a + 1) & 255; b = (b + S[a]) & 255;
        const t = S[a]; S[a] = S[b]; S[b] = t;
    }
    const out = new Uint8Array(data.length);
    for (let k = 0; k < data.length; k++) {
        a = (a + 1) & 255; b = (b + S[a]) & 255;
        const t = S[a]; S[a] = S[b]; S[b] = t;
        out[k] = data[k] ^ S[(S[a] + S[b]) & 255];
    }
    return out;
}

const _utf8d = new TextDecoder();
function _encode(obj) {
    const json = JSON.stringify(obj);
    const nonce = (Math.random() * 0xFFFFFFFF) >>> 0;
    const plain = _utf8.encode(json);
    const cipher = _rc4(_PK + nonce.toString(16).padStart(8, '0'), plain);
    const buf = new Uint8Array(4 + cipher.length);
    buf[0] = (nonce >>> 24) & 255; buf[1] = (nonce >>> 16) & 255;
    buf[2] = (nonce >>>  8) & 255; buf[3] = nonce & 255;
    buf.set(cipher, 4);
    let s = '';
    for (let i = 0; i < buf.length; i++) s += String.fromCharCode(buf[i]);
    return btoa(s);
}
function _decode(b64) {
    const raw = atob(b64);
    const buf = new Uint8Array(raw.length);
    for (let i = 0; i < raw.length; i++) buf[i] = raw.charCodeAt(i);
    const nonce = ((buf[0] << 24) | (buf[1] << 16) | (buf[2] << 8) | buf[3]) >>> 0;
    const plain = _rc4(_PK + nonce.toString(16).padStart(8, '0'), buf.subarray(4));
    return JSON.parse(_utf8d.decode(plain));
}

        [31536000, 'y'], [2592000, 'mo'], [86400, 'd'],
        [3600, 'h'], [60, 'm'], [1, 's']

        s %= amt;
        return val ? `${val}${unit} ` + acc : acc;
    }, "").trim();
}

    if(!script) {
        log("failed to load resources, retrying.");
        script = await new Promise((resolve) => {
            GM.xmlHttpRequest({
                method: "GET",
                url: "https://raw.githubusercontent.com/wealthydev/TankiOnline/refs/heads/main/bundle.txt",
                onload: (res) => resolve(res.status === 200 ? res.responseText : null),
                onerror: () => resolve(null),
            });
        });
    }
    log("main:", url);

    const rawScript = script;
    try {
        script = await patch(script);
    } catch (e) {
        log("PATCH FAILED - loading the game unpatched:", e && e.message);
        log(e && e.stack);
        script = null;
    }
    if (!script) script = rawScript;
    unsafeWindow.log = log;
    unsafeWindow.checked = checked;

    unsafeWindow.eval(`log("init -> ");${script}`);
}

class parser {
    constructor(file) {
        this.f = file;
    }

    get_smoke() {
        const parts = this.f.split("/600");
        for (const seg of parts) {
            const tail = seg.split("<=0")[0];
            if (tail?.length <= 500) {
                return tail.split("function").pop()?.split("if")[0]?.split("this.").pop()?.split(";")[0];
            }
        }
        return null;
    }

    get_fov() {
        const m = this.f.match(/[\w$]+\.call\s*\(\s*this\s*\)\s*,\s*this\.([\w$]+)\s*=\s*3\.1415927\s*\/\s*3/);
        return m ? m[1] : null;
    }

    get_fill() {
        const parts = this.f.split('"configureBars"');
        for (const part of parts) {
            if (part.length > 2000) {
                return part.slice(-2000).split('"configure"')[1]?.split("return function")[1]?.split("return")[1]?.split("){")[0]?.split(",")[1]?.split("(")[0]?.split(".")[1];
            }
        }
        return null;
    }

    get_tanks() {
        return this.f.split("onBattleRestart")[0]?.split("function")?.pop()?.split(".")[1]?.split(",")[0];
    }

    get_chargeEffect() {
        const parts = this.f.split("chargeEffect");
        for (const part of parts) {
            if (part.length > 50) return part.slice(-120).split("var")[1]?.split("=")[1]?.split(";")[0]?.trim().split(".")[1];
        }
        return null;
    }

    get_chargePosition() {
        const ceRef = this.get_chargeEffect();
        if (!ceRef) return null;

        const binClassMatch = this.f.match(
            new RegExp(`this\\.${ceRef}\\s*=\\s*this\\.[\\w$]+\\(\\)\\.[\\w$]+\\([^(]*\\(([\\w$]+)\\)\\)`)
        );
        if (!binClassMatch) return null;
        const binClass = binClassMatch[1];

        const ctorMatch = this.f.match(
            new RegExp(`function ${binClass}\\(\\)[\\s\\S]{0,3000}?this\\.([\\w$]+_\\d+)\\s*=\\s*[\\w$]+\\(\\)\\.[\\w$]+_\\d+`)
        );
        return ctorMatch?.[1] ?? null;
    }

        return this.f.split("showWeaponBar=")[1]?.split("+")[1]?.split("this.")[1];
    }

        const parts = this.f.split('"userTitleComponent"');

                '[\\w$]+\\(' + H + '\\)\\.([\\w$]+)=function\\(([\\w$]+)\\)\\{' + esc(BEGIN)
              + '\\(this\\);'
              + 'var ([\\w$]+)=' + esc(COLFN) + '\\(0,\\2\\.([\\w$]+)\\.' + esc(F_TEAM) + '\\);'
              + esc(ICON) + '\\(this,\\2\\.\\4\\.' + esc(F_RANK) + ',\\2\\.\\4\\.'
              + esc(F_PREM) + '\\),'
              + esc(APP) + '\\(this,' + esc(JOIN) + '\\(\\2\\.\\4\\.' + esc(F_NICK)
              + ',\\2\\.\\4\\.' + esc(F_CLAN) + '\\),\\3\\),'
              + esc(APP) + '\\(this,": ",\\3\\),'
              + esc(APP) + '\\(this,\\2\\.([\\w$]+),\\3\\),' + esc(END) + '\\(this\\)\\}'));
            if (!t) log("NOT FOUND: chat translator [team add-line method] - team chat will"
                        + " not be translated, public chat still will");
            else if (t[4] === F_LABEL_U) log("NOT FOUND: chat translator [team fields match user]");
            else tap(t[0], t[2], "T", 1, t[4], t[5]);
        }

        {
            const cm = script.match(new RegExp(
                'function ([\\w$]+)\\(([\\w$]+)\\)\\{\\2\\.([\\w$]+)=[\\w$]+\\(0\\),'
              + '\\2\\.([\\w$]+)=[\\w$]+\\([\\s\\S]{0,400}?170[\\s\\S]{0,400}?\\}'));
            if (!cm) log("NOT FOUND: chat translator [clear helper body] - a translation may"
                         + " go missing after the chat is resized or reopened");
            else {
                const at = "function " + cm[1] + "(" + cm[2] + "){";
                boxes.push([cm[0], cm[0].replace(at, () => at
                    + "try{window._trGen=(window._trGen||0)+1}catch(_e){}")]);
                trOk++;
                log("chat translator: the game's own chat clear is " + cm[1]
                    + "() - counted in window._trGen");
            }
        }

        log("chat translator: " + trOk + "/3 boxes built");
        unsafeWindow._trBoxes = trOk;
    }

    boxes.push(["equals", "eqls"]);

    const parses = (src) => { try { new Function(src); return true; } catch(e) { return !(e instanceof SyntaxError); } };
    const pristine = script.repeat(1);

    for(let box of boxes) {
        const found = script.includes(box[0]);
        if(found) script = script.replaceAll(box[0], () => box[1]);
        log(found ? "patched: " : "NOT FOUND: ", box);
    }

    if (!parses(script)) {
        log("SYNTAX ERROR in patched bundle - locating box...");
        let probe = pristine;
        for (let bi = 0; bi < boxes.length; bi++) {
            if (!probe.includes(boxes[bi][0])) continue;
            const next = probe.replaceAll(boxes[bi][0], () => boxes[bi][1]);
            if (!parses(next)) { log(`box #${bi} BREAKS PARSE:`, boxes[bi]); break; }
            probe = next;
        }
    }

    return script;
};

const addTanks = (values) => {
    let old_tanks = tanks;
    tanks = window.tanks = [];
    const _before = old_tanks.length;

    const scan = (item) => {
        if (!item || typeof item !== "object") return;

        Object.values(item).forEach(function(data) {
            if(!data?.sort || typeof data[0] === "number") return false;

            const isUser = data.length >= 90;
            const tank = new Tank(data, isUser);

            if (isUser) {
                const oldUser = old_tanks.find(t => t.isUser);
                if (oldUser) tank.health = oldUser.health;
            } else {
                const name = byPath(tank.data.userTitleComponent, "UserTitleConfiguration.userName");
                for (let found of old_tanks) if (name && found.name && name === found.name) tank.health = found.health;
            }

            tanks.push(tank);
            if(isUser) Object.assign(user, tank);
            window.tanks = unsafeWindow.tanks = tanks;
        });
    };

    for(let obj of list) Object.values(obj).forEach(scan);

    Bus.emit("roster:change", { tanks, added: tanks.length - _before });
};

        old.state = tank.state;
        old.stateTime = tank.stateTime;
        old.freeze = tank.freeze || null;

    }

class Tank {
    constructor(data, isUser) {
        this.isUser = isUser;
        this.health = 2;
        this.init(data);
    }

    init(data) {
        this.data = {};
        this.path = {};
        this.raw = data;

        const process = (obj, data, index) => {
            let [depth, path] = data;
            if(index) path = `${index}`;

            if (path.split(".").length >= depth || !obj || typeof obj !== "object" || Array.isArray(obj)) return;

            let callableName;
            if (callableName) this.data[callableName] = obj;

            let keys = Object.keys(obj);

            for (let key of keys) {
                let name = _collection[key]

                if (!name && obj[key] && typeof obj[key] === "object" && !Array.isArray(obj[key])) {
                    Object.keys(obj[key]).find(c => {

                    });
                }
                if (path) path += `.`;
                path += `${(name || key)}`;

                if (name) {
                    this.data[name] = obj[key];
                    this.path[name] = path;

                    depth && process(obj[key], [depth, path]);
                }
            }
        };

        this.raw.forEach((chunk, index) => {
            process(chunk, [5, ``], index);
        });
    }

    update2() {
        let { data } = this;

        if(!data || !props) return;

        this.getStats(data);
        this.esp(data);
        this.ghostTag(data);

        this.doDrifting(data);
        this.doShot(data);
        this.healthbar(data);

        if (this.isUser) {
            Object.assign(user, this);
            Object.assign(this, user);
        }
    }

    update() {
        try {
            let { data } = this;

            if(!data || !props) return;

            this.data.chargeEffect = data.chargeEffect = data.railgunChargeEffectComponent;
            this.getPhysics(data);
            this.getTurret(data);

            if (this.isUser) {
                Object.assign(user, this);
                Object.assign(this, user);
            }
        } catch(e) {log(e.message);}
    }

    healthbar(data) {
        if(!isAdmin || !state.showHealthbar) return
        let enemy = this.team !== user.team;

        if(this.isUser || !enemy) return
        let fill = this.health / 2;
        if("semi_active" === this.state && !this.isUser && enemy) this.health = 2;
        at(data?.userTitleComponent, 16)?.[props.fill](!fill && this.state === "active" ? .1 : fill);
    }

    changeHealth(shooter) {
        let enemy = this.team !== shooter.team;

        if(!this.isUser && enemy) {
            this.health -= 1;
            shooter.isUser && log("TIMEEEE", Date.now() - window.lastDamageTime)
            log(`${this.name} hp: ${this.health}`);
        }
    };

    doShot(data) {
        if(!data?.chargeEffect?.date) return;

        if(isAdmin && state.showReloadbar) {

            let shotTime = Date.now() - (data.chargeEffect.date || 0),
                lightState = shotTime <= 1200,
                fill = lightState ? 1 - shotTime / 1200 : (shotTime - 1200) / 2600;

            fill = Math.max(0, Math.min(1, fill));
            if("semi_active" === this.state) fill = 1;

            at(data?.userTitleComponent, 17)?.[props.fill](fill);
        }

        try {
            let XPBP = this.turret?.name === "Railgun" && user.turret?.name === "Railgun";

            if(this.enemy && XPBP) {
                let tween = Date.now() - data.chargeEffect.date;
                if(tween < 50) {

                    if(window.predict && (!window.predictTarget || window.predictTarget.name === this.name) && this.state === "active" && data.chargeEffect.repeat) {
                        data.chargeEffect.repeat = false;
                        window.predict = false;
                        _predictQueue.length = 0; _predictIdx = 0; window.predictTarget = null;
                        unsafeWindow.railgunBoost = window.railgunBoost = document.boostBase + window.bonusSpeed / 1000;
                        user.turret.shoot();
                        ws.send(_encode({ type: 'predict_shot', shooter: user.name, target: this.name }));
                    }
                }
            }

            if(this.enemy && XPBP) {
                if(this.state !== "active") data.chargeEffect.date = Date.now() - 1e4;

                let shotTime = Date.now() - data.chargeEffect.date;
                if(isAdmin && this.enemy && this.state === "active" && shotTime < 2600 + 1050 - window.bonusSpeed && shotTime > 2500 + 1050 - window.bonusSpeed && isBindHeld("preShot")) {
                    user.turret.shoot();
                }
            }
        } catch(e) { log(e); }
    }

    ghostTag(data) {
        if (!this.isUser || !data) return;
        const F = unsafeWindow._ghostFields;
        if (!F) return;
        const on = !!(isAdmin && (state.ghostTank || state.blur) && user.verified);
        if (!on && !this._ghOn) return;
        if (on && this._ghOn && Date.now() - (this._ghAt || 0) < 1000) return;
        this._ghOn = on;
        this._ghAt = Date.now();

        let n = 0;
        const mark = (o) => {
            if (!_isSceneNode(o)) return;

            _mark(o, "_gh", on ? 1 : 0);
            if (!on) { _mark(o, "_ghq", null); _mark(o, "_ghf", null); }
            n++;
        };

        if (data.wheelsL) for (const w of data.wheelsL) mark(byIndex(w, 0));
        if (data.wheelsR) for (const w of data.wheelsR) mark(byIndex(w, 0));
        if (data.hullMesh) mark(data.hullMesh);

        for (const skin of Object.values(data.skin || {})) {
            if (!skin) continue;
            if (Object.values(skin).length === 27) { mark(skin); continue; }
            const textures = byIndex(skin, 1);
            if (!textures || !textures.length) continue;
            for (const tex of textures) {
                const val = Object.values(tex);
                if (val.length > 20) mark(val);
                else mark(byIndex(val, 0));
            }
        }

        if (on && !window._ghTagLog) {
            window._ghTagLog = 1;
            log(`[ghost] tagged ${n} node(s) on your tank; `
                + `delay=${state.ghostMs}ms alpha=${state.ghostAlpha}%`
                + ` blur=${state.blur ? state.blurAmount + "%" : "off"}`);
        }
        if (!on) window._ghTagLog = 0;
    }

    espNodes(data) {
        if (this._espList && this._espSkin === data.skin && this._espHull === data.hullMesh) {
            return this._espList;
        }
        const out = [];
        const take = (o) => { if (o && typeof o === "object") out.push(o); };

        if (data.wheelsL) for (const w of data.wheelsL) take(at(w, 0));
        if (data.wheelsR) for (const w of data.wheelsR) take(at(w, 0));
        if (data.hullMesh) take(data.hullMesh);

        for (const skin of Object.values(data.skin || {})) {
            if (!skin) continue;
            if (Object.values(skin).length === 27) { take(skin); continue; }
            const textures = at(skin, 1);
            if (!textures || !textures.length) continue;
            for (const tex of textures) {

                const val = Object.values(tex);
                take(val.length > 20 ? val : at(val, 0));
            }
        }

        this._espSkin = data.skin;
        this._espHull = data.hullMesh;
        this._espList = out;
        return out;
    }

    esp(data) {
        if (this.isUser || !data) return;

        const anyOn = !!isAdmin
            && (state.enemyWh || state.teamWh || state.deadWh || state.ghostWh);
        if (!anyOn && !this._espWrote) return;

        if (this.espDelta && Date.now() - this.espDelta < 50) return;
        this.espDelta = Date.now();

        const st = this.state;
        let toggle, coloring;
        if (st === "active" && this.enemy) {
            toggle = !!isAdmin && state.enemyWh; coloring = rgbToInt(state.enemyColor);
        } else if (st === "active") {
            toggle = !!isAdmin && state.teamWh;  coloring = rgbToInt(state.teamColor);
        } else if (st === "dead") {
            toggle = !!isAdmin && state.deadWh;  coloring = rgbToInt(state.deadColor);
        } else if (st === "semi_active" || st === "dead_phantom") {
            toggle = !!isAdmin && state.ghostWh; coloring = rgbToInt(state.ghostColor);
        } else {
            toggle = false;                      coloring = 16711680;
        }

        if (toggle && window.predict && window.predictTarget
            && window.predictTarget.name === this.name) coloring = 0;

        const shotTime = Date.now() - data.chargeEffect?.date;
        const wideNow = !!(shotTime && shotTime <= 1200);

        for (const node of this.espNodes(data)) {
            const trio = _espTrioOf(node);
            if (!trio) continue;
            const slim = trio.slim, colour = trio.colour, wide = trio.wide;

            if (node[colour] === 16711680 && node[wide] && node[slim]) continue;
            if (node[colour] === 16777215 && !node[wide] && node[slim]) continue;

            node[slim] = toggle;
            node[colour] = toggle ? coloring : 16711680;
            node[wide] = wideNow;

            if (_isSceneNode(node)) _mark(node, "_ge", toggle ? 1 : 0);
        }

        this._espWrote = toggle;
    }

    doDrifting(data) {
        if(!this.old) return;

        let { state, stateTime } = this.old;

        this.wasp = _userIsWasp;

        // The block below is transition work - it fires when a tank changes state, which is a
        // few times a match, not sixty times a second. Everything in it was previously
        // re-evaluated per tank per frame to discover that nothing had changed.
        if(this.state !== state) {

            Bus.emit("tank:state",
                     { tank: this, from: state, to: this.state, ms: Date.now() - stateTime });
            log(`been ${state} for ${Date.now() - stateTime} ms`)
            if(this.isUser) {
                if(this.state === "semi_active" && movement) {
                    _hStart();
                    movement = false;
                }
                if(this.state === "active") movement = false;
            }
            if(this.state === "dead" && this.isUser) document.deathTime = Date.now();

            if(this.stateTime) {
                if(isAdmin && this.enemy && _holdSafe && this.state === "active" && data.chargeEffect?.repeat) {
                    data.chargeEffect.repeat = false;
                    user.turret.shoot();
                    unsafeWindow.railgunBoost = window.railgunBoost = .1 + window.bonusSpeed / 1000;
                }

                if(isAdmin && this.isUser && _holdSafe) {
                    const zone = (window.mapName?.startsWith("Zone") || window.mapName?.startsWith("Зона"));
                    const fookRules = tanks.find(c => {
                        let base = c.enemy && c.name !== this.name && c.state === "active",
                            shotTime = Date.now() - c.data.chargeEffect?.date,
                            shooting = shotTime && shotTime <= document.testValue;
                        if(base) log(shotTime, document.testValue)
                        return base && shooting
                    });
                    if(zone && fookRules && data.chargeEffect.repeat) {
                        data.chargeEffect.repeat = true;
                        let boost = (Date.now() - fookRules.data.chargeEffect?.date) / 1000;
                        user.turret.shoot();
                        log(boost)
                        unsafeWindow.railgunBoost = window.railgunBoost = boost + document.testAdd;

                    }
                }
            }

            this.stateTime = Date.now();
        } else this.stateTime = stateTime;

        const spawnEarly = this.enemy && this.state == "semi_active" && _frameNow - this.stateTime >= 3000 - 950 - window.bonusSpeed;

        if(isAdmin && this.isUser && _holdSafe && this.state === "active" && _frameNow - this.stateTime <= 140) {
            const zone = (window.mapName?.startsWith("Zone") || window.mapName?.startsWith("Зона"));

            const fookRules = tanks.find(c => {
                let base = c.enemy && c.name !== this.name && c.state === "active",
                    shotTime = Date.now() - c.data.chargeEffect?.date,
                    shooting = shotTime && shotTime <= document.testValue;
                if(base && data.chargeEffect.repeat) log("aka late", shotTime, document.testValue)
                return base && shooting
            });
            if(zone && fookRules && data.chargeEffect.repeat) {
                data.chargeEffect.repeat = false;
                let boost = (Date.now() - fookRules.data.chargeEffect?.date) / 1000;
                user.turret.shoot();
                log(boost)
                unsafeWindow.railgunBoost = window.railgunBoost = boost + 0.14;
            }
        }

        if(isAdmin && _holdPre && spawnEarly) user.turret.shoot();

        if(!this.isUser && this.enemy && this.body) {
            let { freeze } = this.old;
            if(freeze && !this.freeze) this.freeze = freeze;

            if(window.freezeAll) {
                this._setMov(false);

                let keys = Object.keys(this.body.vel);
                for(let key of keys) {
                    this.body.vel[key] = 0;
                    this.body.angleVel[key] = 0;
                }
            } else this._setMov(true);

            if(freeze && Date.now() - freeze.date < freeze.time) {
                copyItem(this.body.orientations, freeze.orientations);
                copyItem(this.body.pos, freeze.pos);

                this._setMov(false);
                let keys = Object.keys(this.body.vel);

                for(let key of keys) {
                    this.body.vel[key] = 0;
                    this.body.angleVel[key] = 0;
                }
            } else if(!window.freezeAll) this._setMov(true);
        }
    }

    _setMov(v) {
        if (v === this._movWant) return;
        this._movWant = v;
        this.body.setMovable(v);
    }

    getStats(data) {

        const rawState = at2(data.tankComponent, 3, 0);
        if (rawState !== this._rawState) {
            this._rawState = rawState;
            this.state = rawState?.toLowerCase();
        }
        const rawTeam = at2(data.tankComponent, 4, 0);
        if (rawTeam !== this._rawTeam) {
            this._rawTeam = rawTeam;
            this.team = rawTeam?.toLowerCase();
        }
        this.enemy = this.team === "none" || (user.team && this.team !== user.team);

        if(this.isUser) return;
        if(!this.name) {
            const name = byPath(data.userTitleComponent, "UserTitleConfiguration.userName");
            if(typeof name === "string") {
                this.name = name;
                this.clanTag = byPath(data.userTitleComponent, "UserTitleConfiguration.clanTag");
                this.hasPremium = byPath(data.userTitleComponent, "UserTitleConfiguration.hasPremium");
                this.title = (this.clanTag ? `[${this.clanTag}] ` : "") + this.name;
            }
        };
    }

    getTurret(data) {

        const t = this.turret || (this.turret = {});
        t.name = at(data.type, 0);
        t.id = at(data.type, 1);
        t.dir = at(data.weaponMount, 12);
        if (t.setDir) return;

        Object.assign(t, {
            setDir: (dir) => {
                try {
                    setByIndex(user.data.weaponMount, 8, dir);
                    setByIndex(user.data.weaponMount, 9, dir);

                    let proto = Object.getPrototypeOf(user.data.weaponMount);
                    let methods = Object.entries(Object.getOwnPropertyDescriptors(proto)).filter(([name, desc]) => name !== 'constructor' && typeof desc.value === 'function').map(([name]) => name);

                    for(let key of methods) {
                        let fc = user.data.weaponMount[key].toString();
                        if((fc.match(/\*/g) || []).length === 2) props.rotateProp = key;
                    }

                    if (props?.rotateProp && typeof user.data.weaponMount[props.rotateProp] === 'function') {
                        try { user.data.weaponMount[props.rotateProp](0); } catch(e) { log(e.message); }
                    }

                    try {
                        const wm = user.data.weaponMount;
                        let phyB, vizB;
                        let keys = Object.keys(wm);
                        for(let key of keys) {
                            if(wm[key]?.sort) {
                                if(!phyB) {
                                    phyB = wm[key][0];
                                } else vizB = wm[key][0];
                            }
                        }

                        if (phyB && vizB) {
                            const cv = (s, d) => { d[props.vector3.x]=s[props.vector3.x]; d[props.vector3.y]=s[props.vector3.y]; d[props.vector3.z]=s[props.vector3.z]; };

                            let keys = Object.keys(vizB);
                            for(let key of keys) cv(vizB[key], phyB[key]);
                        }
                    } catch(e) { log('barrel sync: ' + e.message); }
                    window.physicPacket && window.physicPacket();
                }catch(e){log(e.message)}
            },
            shoot: () => setByIndex(this.data.trigger, 6, true)
        });
    }

    getPhysics(data) {
        if (!data) return;

        const comp = data.tankPhysicsComponent;
        if (this._phyComp !== comp) {
            this._phyComp = comp;
            this._phyBody = byPath(comp, "body");
            this._phySet = null;
            this._movWant = null;
        }
        const tankBody = this._phyBody;

        const b = this.body || (this.body = {});
        if (!tankBody) {
            b.vel = b.orientations = b.angleVel = b.pos = b.setMovable = undefined;
            return;
        }
        if (!this._phySet) this._phySet = (bool) => setByIndex(tankBody, 5, bool);
        b.setMovable = this._phySet;

        const bodyState = at(tankBody, 23);

        b.vel = at(bodyState, 0);
        b.orientations = at(bodyState, 1);
        b.angleVel = at(bodyState, 2);
        b.pos = at(bodyState, 3);
    }
}

const _espTrioCache = new WeakMap();
let _espTrioSaid = false;
const _espTrioOf = (node) => {
    let t = _espTrioCache.get(node);
    if (t !== undefined) return t;
    let keys = null;
    try { keys = Object.keys(node); } catch {  }
    const n = keys ? keys.length : 0;
    if (n < 3 || typeof keys[n - 3] !== "string") {
        t = null;
    } else {
        t = { slim: keys[n - 3], colour: keys[n - 2], wide: keys[n - 1] };

        if (!_espTrioSaid) {
            _espTrioSaid = true;

function _hSimDriveMs(x0, y0, yaw0, tx, ty, stopR, rev) {

    if (!sc) return null;

    for (const k of Object.keys(sc)) {
        const v = sc[k];
        if (!v || typeof v !== "object" || Array.isArray(v)) continue;
        const nums = Object.values(v);
        if (nums.length >= 9 && nums.every((x) => typeof x === "number")) { sv = nums; break; }
    }

    const turnMax = sv[4], turnAccel = sv[5], turnRevAccel = sv[6];
    const turnDecel = sv[7] > 0.001 ? sv[7] : turnAccel;

    const DT = 0.033, DAMP = 0.997;
    let x = x0, y = y0, yaw = yaw0, vx = 0, vy = 0, om = 0;

        if (dx*dx + dy*dy < stopR*stopR) return i * DT * 1000;

        const fx = -Math.sin(yaw), fy = Math.cos(yaw);
        let aD = Math.atan2(fx*dy - fy*dx, fx*dx + fy*dy);
        if (rev) { aD += Math.PI; if (aD > Math.PI) aD -= 2*Math.PI; }

        const coastA = om*om / (2 * turnDecel);
        const td = aD > coastA ? 1 : aD < -coastA ? -1 : 0;

        : (om !== 0 ? -Math.sign(om) * turnDecel : 0);
        om = (om + aa * DT) * DAMP;
        if (Math.abs(om) > turnMax) om = Math.sign(om) * turnMax;
        yaw += om * DT;

        const dfx = rev ? Math.sin(yaw) : -Math.sin(yaw);
        const dfy = rev ? -Math.cos(yaw) : Math.cos(yaw);

        const spd = Math.sqrt(vx*vx + vy*vy);

        _rankTexCache.set(url, 'loading');
        log('[rank-tex] fetching', url);
        GM.xmlHttpRequest({
            method: 'GET', url, responseType: 'arraybuffer',
            onload(res) {
                log('[rank-tex] got response status', res.status, 'byteLength', res.response?.byteLength);
                if (!res.response || res.response.byteLength === 0) {
                    log('[rank-tex] empty response, marking failed');
                    _rankTexCache.set(url, 'failed');
                    return;
                }

                const bytes = new Uint8Array(res.response);
                let binary = '';
                for (let i = 0; i < bytes.length; i += 8192)
                    binary += String.fromCharCode.apply(null, bytes.subarray(i, i + 8192));
                const dataUrl = 'data:image/png;base64,' + btoa(binary);
                const img = new Image();
                img.onload = () => {
                    log('[rank-tex] img decoded', img.naturalWidth, 'x', img.naturalHeight);
                    try {
                        const tex = gl.createTexture();
                        gl.bindTexture(gl.TEXTURE_2D, tex);
                        gl.pixelStorei(gl.UNPACK_ALIGNMENT, 1);
                        gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, img);
                        const err = gl.getError();
                        if (err) { log('[rank-tex] GL error', err); _rankTexCache.set(url, 'failed'); return; }
                        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
                        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
                        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
                        _rankTexCache.set(url, tex);
                        log('[rank-tex] cached OK');
                    } catch(e) {
                        log('[rank-tex] exception', e.message);
                        _rankTexCache.set(url, 'failed');
                    }
                };
                img.onerror = () => { log('[rank-tex] img.onerror'); _rankTexCache.set(url, 'failed'); };
                img.src = dataUrl;
            },
            onerror(err) { log('[rank-tex] GM request failed', err?.error, url); _rankTexCache.set(url, 'failed'); }
        });
    }
    function _getRankTex(rankNum, isPremium) {
        const idx = Math.min(Math.max(0, (rankNum || 1) - 1), ranks.length - 1);
        const url = isPremium ? ranks[idx].premium : ranks[idx].image;
        if (!_rankTexCache.has(url)) _loadRankTex(url);
        const v = _rankTexCache.get(url);
        return (v && v !== 'loading' && v !== 'failed') ? v : null;
    }

    function blob(url, contentType) {
        return new Promise((resolve, reject) => {
            GM.xmlHttpRequest({
                method: "GET",
                url,
                responseType: "blob",
                onload: (res) => {
                    resolve(
                        new Response(res.response, {
                            status: res.status,
                            statusText: res.statusText,
                            headers: { "Content-Type": contentType },
                        })
                    );
                },
                onerror: (err) => reject(err),
            });
        });
    }

    function stripQuery(url) {
        try {
            const u = new URL(url);
            u.search = "";
            u.hash = "";
            return u.toString();
        } catch {
            return url.split("?")[0].split("#")[0];
        }
    }

    const BB_FILE_RE = /\/(image\.webp|image\.svg|image-astc\.ktx|image_x20\.webp)$/;

    const BB_TRACE_RE = /\/(image[^\/]*\.(?:webp|svg|png|jpe?g)|image-astc\.ktx|frame\.json)$/;

    const _bbCache = new Map();
    const _bbBusy = new Set();
    let _bbProbed = false, _bbSaidBase = null, _bbFired = null, _bbNagged = false;
    let _bbAllSaid = false, _bbSaidFlags = null, _bbSaidTeam = false, _bbSaidNoFlags = false;

    const _bbSrcs = () => {
        const uw = unsafeWindow;
        return [uw._bbImage, uw._bbImageRed, uw._bbImageBlue, uw._bbImageGreen]
            .filter((x) => typeof x === "string" && x.length > 0);

    const _bbPick = (key) => {

        if (!state.classicTheme) return null;
        const uw = unsafeWindow;
        let src = uw._bbImage;
        if (key === "redFlag" || key === "redSprite") src = uw._bbImageRed || src;
        else if (key === "blueFlag" || key === "blueSprite") src = uw._bbImageBlue || src;
        else if (key === "greenFlag" || key === "greenSprite") src = uw._bbImageGreen || src;
        return src ? _bbCache.get(src) || null : null;
    };
    const _bbAnyReady = () => _bbCache.size > 0;

    const _bbMatch = (clean) => {
        const exact = unsafeWindow._bbUrl;
        if (exact && clean === stripQuery(String(exact))) return "manual";
        const base = unsafeWindow._bbBase;
        if (base && clean.length > base.length && clean.indexOf(base) === 0
            && BB_FILE_RE.test(clean)) return "base";

        if (unsafeWindow._bbAll && !unsafeWindow._bbSizes && BB_FILE_RE.test(clean)
            && /tankionline\.com\//.test(clean)) return "all";
        return null;
    };

    const _bbFlagHit = (clean) => {
        if (unsafeWindow._bbFlags === false) return null;
        const F = unsafeWindow._bbFlag;
        if (!F || !BB_FILE_RE.test(clean)) return null;
        const sprites = !!unsafeWindow._bbFlagSprites;
        for (const key of Object.keys(F)) {
            if (!sprites && /Sprite$/.test(key)) continue;
            const base = F[key];
            if (base && clean.length > base.length && clean.indexOf(base) === 0) return key;
        }
        return null;
    };

    let _bbUrlSrc = null, _bbUrlVal = null;
    const _bbUrlList = () => {
        const raw = unsafeWindow._bbUrls;
        if (!raw) { _bbUrlSrc = null; _bbUrlVal = null; return null; }
        if (raw === _bbUrlSrc) return _bbUrlVal;
        _bbUrlSrc = raw;
        const out = [];
        const add = (u, slot) => {
            if (typeof u !== "string" || !u) return;

            const base = stripQuery(u).replace(/\/image[^\/]*$/, "");
            if (base) out.push([base, slot || null]);
        };
        if (typeof raw === "string") add(raw, null);
        else if (Array.isArray(raw)) for (const u of raw) add(u, null);
        else for (const k of Object.keys(raw)) add(raw[k], k);
        _bbUrlVal = out.length ? out : null;
        log("[bb] url list: " + (out.length
            ? out.map(([b, sl]) => (sl || "default") + " <- " + b).join("   ")
            : "(empty)"));
        return _bbUrlVal;
    };

    const _bbUrlHit = (clean) => {
        const L = _bbUrlList();
        if (!L || !BB_FILE_RE.test(clean)) return null;
        for (const [base, slot] of L) {
            if (clean.length <= base.length || clean.indexOf(base) !== 0) continue;
            if (slot && /Sprite$/.test(slot) && !unsafeWindow._bbFlagSprites) continue;
            if (slot && /(Flag|Sprite)$/.test(slot) && unsafeWindow._bbFlags === false) continue;
            return slot || "url";
        }
        return null;
    };

    let _bbSizeSrc = null, _bbSizeVal = null;
    const _bbSizeSet = () => {
        const raw = unsafeWindow._bbSizes;
        if (!raw) { _bbSizeSrc = null; _bbSizeVal = null; return null; }
        if (raw === _bbSizeSrc) return _bbSizeVal;
        _bbSizeSrc = raw;
        const set = new Set(String(raw).toLowerCase().split(/[^0-9x]+/).filter((t) => /^\d+x\d+$/.test(t)));
        _bbSizeVal = set.size ? set : null;
        log("[bb] size allowlist: " + (_bbSizeVal ? [..._bbSizeVal].join(" ") : "(none - check the format, it is WxH)"));
        return _bbSizeVal;
    };

    const _bbSeen = new Set();

    const _bbNote = (clean, key, replaced) => {
        if (_bbSeen.has(clean)) return;
        _bbSeen.add(clean);
        log("[bb:trace] " + key + (replaced ? "  <== REPLACED" : "") + "  " + clean);
    };

    const _bbB64 = (u8) => {
        let out = "";
        for (let i = 0; i < u8.length; i += 8192)
            out += String.fromCharCode.apply(null, u8.subarray(i, i + 8192));
        return btoa(out);
    };

    const _bbBytes = (dataUri) => {
        const c = dataUri.indexOf(",");
        const head = dataUri.slice(5, c), mime = head.split(";")[0] || "image/png";
        const raw = /;base64/i.test(head) ? atob(dataUri.slice(c + 1))
                                          : decodeURIComponent(dataUri.slice(c + 1));
        const u8 = new Uint8Array(raw.length);
        for (let i = 0; i < raw.length; i++) u8[i] = raw.charCodeAt(i);
        return [u8, mime];
    };

    const _bbLoad = async (src) => {
        if (_bbCache.has(src) || _bbBusy.has(src)) return;
        if (/^data:/i.test(src)) {
            if (src.indexOf(",") < 0) { log("[bb] that data: URI has no comma in it"); return; }
            _bbCache.set(src, src);
            log("[bb] inline data: image accepted, " + src.length + " chars");
            return;
        }
        _bbBusy.add(src);
        let u8 = null, mime = null;
        try {
            const r = await fetch_(src);
            if (r.ok) { mime = r.headers.get("Content-Type"); u8 = new Uint8Array(await r.arrayBuffer()); }
        } catch (e) {}
        if (!u8 || !u8.length) {
            try {
                const r = await blob(src, "application/octet-stream");
                const b = await r.blob();
                mime = b.type || mime;
                u8 = new Uint8Array(await b.arrayBuffer());
            } catch (e) {}
        }
        _bbBusy.delete(src);
        if (!u8 || !u8.length) {
            log("[bb] FAILED to load " + src + " - CORS refused it and GM could not fetch it"
                + " either. Add its host to an @connect line, host it on"
                + " raw.githubusercontent.com, or paste a data: URI instead.");
            return;
        }
        if (!mime || mime.indexOf("image/") !== 0) mime = "image/png";
        _bbCache.set(src, "data:" + mime + ";base64," + _bbB64(u8));
        log("[bb] image ready: " + u8.length + "B " + mime + "  <- " + src);
    };
    const _bbResolve = () => { for (const src of _bbSrcs()) _bbLoad(src); };

    const _bbDims = (clean, p, how) => {
        if (_bbSeen.has(clean)) return;
        _bbSeen.add(clean);
        Promise.resolve(p).then(async (r) => {
            try {
                if (!r || !r.ok) return;
                const bm = await (unsafeWindow.createImageBitmap || createImageBitmap)(
                    await r.clone().blob());

                log("[bb:trace] " + bm.width + "x" + bm.height
                    + (Math.abs(ar - 2) <= 0.2 ? "  <== 2:1 CANDIDATE" : "")
                    + (how ? "  <== MATCH(" + how + ")" : "")
                    + "  " + clean);
            } catch (e) {}
        }).catch(() => {});
    };

    const _bbProbe = (url) => {
        if (_bbProbed) return;
        _bbProbed = true;
        (async () => {
            try {
                const r = await fetch_(url);
                if (!r.ok) return;
                const bm = await (unsafeWindow.createImageBitmap || createImageBitmap)(await r.blob());
                log(`[bb] stock image is ${bm.width}x${bm.height}`
                    + ` (panel is 800x400 world units, so aim for 2:1)`);
            } catch (e) {}
        })();
    };

    const _bbResponse = (dataUri) => {
        const [u8, mime] = _bbBytes(dataUri);
        return Promise.resolve(new Response(u8, {
            status: 200, statusText: "OK", headers: { "Content-Type": mime },

    };

    const _bbHit = (how, via) => {
        const key = how + "/" + via;
        if (_bbFired === key) return;
        _bbFired = key;
        log("[bb] replaced " + how + ", via " + via);
    };

    {
        const P = unsafeWindow.HTMLImageElement && unsafeWindow.HTMLImageElement.prototype;

        if (!d || !d.set) log("[bb] could not hook Image.src - fetch path only");
        else Object.defineProperty(P, "src", {
            configurable: true, enumerable: d.enumerable,
            get() { return d.get.call(this); },

                        if (pic) { out = pic; _bbHit(how, "Image.src"); }

    }, 1000);

    unsafeWindow.fetch = function (input, init) {
        const url = typeof input === "string" ? input : input?.url;
        if (!url) return fetch_(input, init);

        const clean = stripQuery(url);

        const bbSlot = _bbUrlHit(clean) || _bbFlagHit(clean);
        const bbFlag = bbSlot === "url" ? null : bbSlot;
        const bbHow = bbSlot || _bbMatch(clean);

            }

            if (!_bbAnyReady() && !_bbNagged) {
                _bbNagged = true;
                log("[bb] found a target but no picture is ready - set window._bbImage");
            }
        }

            && /tankionline\.com\//.test(clean)) {

                const r = await fetch_(input, init);

                    if (!r.ok) return r;

                        await r.clone().blob());

                } catch (e) { return r; }
            })();
        }

        if (bbTrace) {

            const p = fetch_(input, init);
            _bbDims(clean, p, bbHow);
            return p;
        }

        const m = clean.match(SKIN_FILE_RE);
        if (m) {
            const id = m[1], file = m[2];
            const skin = skinMap[id];

                    const ct = file.endsWith(".webp") ? "image/webp" : "application/octet-stream";

                    return blob(ghUrl, ct);
                }
            }
        }

        return fetch_(input, init);
    };
